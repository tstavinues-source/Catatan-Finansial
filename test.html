// --- FUNGSI PENCUCI GAMBAR (PREPROCESSING) --- // Mengubah foto HP biasa menjadi hitam-putih pekat ala mesin scanner async function cuciGambarUntukOCR(base64Str) { return new Promise((resolve) => { const img = new Image(); img.onload = () => { const canvas = document.createElement('canvas'); const ctx = canvas.getContext('2d'); // 1. Perbesar gambar 2x lipat agar huruf kecil tidak pecah saat dibaca const scale = 2; canvas.width = img.width * scale; canvas.height = img.height * scale; ctx.drawImage(img, 0, 0, canvas.width, canvas.height); // 2. Ambil data piksel const imgData = ctx.getImageData(0, 0, canvas.width, canvas.height); const data = imgData.data; // 3. Proses Thresholding (Hitam/Putih Ekstrem) for (let i = 0; i < data.length; i += 4) { const r = data[i]; const g = data[i+1]; const b = data[i+2]; // Rumus mengubah ke Grayscale const grayscale = 0.299 * r + 0.587 * g + 0.114 * b; // Jika kecerahan di bawah 130 (gelap/tinta), jadikan hitam pekat (0) // Jika di atas 130 (kertas/cahaya), jadikan putih bersih (255) const threshold = 130; const color = grayscale < threshold ? 0 : 255; data[i] = data[i+1] = data[i+2] = color; // Terapkan RGB baru } ctx.putImageData(imgData, 0, 0); // Kembalikan gambar yang sudah "dicuci" resolve(canvas.toDataURL('image/jpeg', 1.0)); }; img.src = base64Str; }); } // --- 3. FUNGSI INTI: TESSERACT + GROQ --- async function eksekusiAIPipeline(userPrompt, imageStr, groqKey) { try { let teksMentahStruk = ""; // TAHAP 0: CUCI GAMBAR SEBELUM DIBACA uiLog("🧹 Mencuci gambar (Binarization & Upscaling) agar tulisan struk lebih jelas...", "warn"); const cleanImageStr = await cuciGambarUntukOCR(imageStr); // TAHAP 1: MATA (TESSERACT OCR OFFLINE) uiLog("⚙️ Menyalakan mesin Tesseract.js untuk membaca gambar yang sudah dicuci...", "warn"); const worker = await Tesseract.createWorker('ind+eng', 1, { logger: m => { if(m.status === 'recognizing text' && m.progress % 0.2 < 0.01) { uiLog(`Proses baca: ${Math.round(m.progress * 100)}%`); } } }); uiLog("Mengeja tulisan...", "warn"); // PENTING: Gunakan gambar yang sudah dicuci (cleanImageStr) const ret = await worker.recognize(cleanImageStr); teksMentahStruk = ret.data.text; await worker.terminate(); uiLog("--- HASIL SCAN MENTAH TESSERACT ---", "info"); uiLog(teksMentahStruk, "info"); if(teksMentahStruk.trim() === '') { throw new Error("Tesseract tetap tidak menemukan teks. Coba foto di tempat yang lebih terang."); } // TAHAP 2: OTAK (GROQ AI API) uiLog("🧠 Mengirim teks mentah ke Groq AI (Llama 3.3) untuk dirapikan menjadi JSON...", "warn"); const systemPrompt = `Anda adalah sistem parser data keuangan tingkat lanjut. Berikut adalah hasil scan OCR (Optical Character Recognition) dari sebuah struk belanja. Karena ini dari OCR, teksnya pasti berantakan, terpotong, atau ada salah eja. Tugas Anda: Analisis teks tersebut, tebak maksud sebenarnya dari barang-barang tersebut, dan ekstrak menjadi format JSON murni. Wajib gunakan struktur JSON ini persis: { "total_belanja": angka_murni_tanpa_simbol, "toko": "nama toko", "daftar_barang": ["nama barang 1", "nama barang 2"] } ATURAN MUTLAK: Hanya berikan respons JSON mentah! Dilarang menambahkan penjelasan, sapaan, atau markdown backticks (\`\`\`json).`; const finalPrompt = `[TEKS STRUK MENTAH HASIL OCR]\n${teksMentahStruk}\n\n[INSTRUKSI TAMBAHAN]\n${userPrompt}`; const payload = { model: "llama-3.3-70b-versatile", messages: [ { role: "system", content: systemPrompt }, { role: "user", content: finalPrompt } ], temperature: 0.1 }; const response = await fetch("https://api.groq.com/openai/v1/chat/completions", { method: "POST", headers: { "Authorization": `Bearer ${groqKey}`, "Content-Type": "application/json" }, body: JSON.stringify(payload) }); if (!response.ok) { const errDetail = await response.json().catch(()=>({})); throw new Error(`Groq API Error: ${response.status} - ${errDetail.error?.message || ''}`); } const result = await response.json(); let outputAI = result.choices[0].message.content.trim(); uiLog("--- 🚀 HASIL AKHIR GROQ (JSON RAPI) ---", "success"); uiLog(outputAI, "success"); } catch (err) { uiLog(err.message, "error"); } finally { const btn = document.getElementById('btn-test'); btn.disabled = false; btn.innerHTML = ' Eksekusi AI Pipeline'; } }
